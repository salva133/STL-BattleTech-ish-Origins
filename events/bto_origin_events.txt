# Stellar Union/Clan Confederation Setup
# Stellar Union = the Inner Sphere: leader (centre) + 5 inner realms in a ring.
# Clan Confederation = leader + 6 clans in a sealed cluster beyond the galactic rim.

namespace = bto_origin

country_event = {
	id = bto_origin.55
	hide_window = yes
	is_triggered_only = yes

	immediate = {
		# Spawn rival empire with opposing BTO origin (guard prevents recursion)
		if = {
			limit = { NOT = { has_country_flag = bto_is_rival_empire } }
			if = {
				limit = { has_origin = origin_clan_confederation }
				bto_spawn_rival_empire = {
					ORIGIN = origin_stellar_union
					AUTH = auth_imperial
					CIVIC1 = civic_aristocratic_elite
					CIVIC2 = civic_distinguished_admiralty
				}
			}
			if = {
				limit = { has_origin = origin_stellar_union }
				bto_spawn_rival_empire = {
					ORIGIN = origin_clan_confederation
					AUTH = auth_oligarchic
					CIVIC1 = civic_warrior_culture
					CIVIC2 = civic_aristocratic_elite
				}
			}
		}

		set_country_flag = first_contact_event
		set_country_flag = Story2 # Already encountered spacefarers
		give_technology = { tech = tech_federation_code message = no }

		# Inner Sphere: the leader is the centre; 5 great houses ring around it.
		if = {
			limit = { has_origin = origin_stellar_union }
			capital_scope = {
				solar_system = {
					bto_spawn_ring_home = { NUM = 1 ANGLE = 0 }
					bto_spawn_ring_home = { NUM = 2 ANGLE = 72 }
					bto_spawn_ring_home = { NUM = 3 ANGLE = 144 }
					bto_spawn_ring_home = { NUM = 4 ANGLE = 216 }
					bto_spawn_ring_home = { NUM = 5 ANGLE = 288 }
				}
			}
		}

		# Clans: 6 homes in a sealed cluster beyond the rim, linked by a gateway.
		if = {
			limit = { has_origin = origin_clan_confederation }
			bto_spawn_cluster_anchor = { NUM = 1 }
			bto_spawn_cluster_home = { NUM = 2 ANGLE = 60 }
			bto_spawn_cluster_home = { NUM = 3 ANGLE = 120 }
			bto_spawn_cluster_home = { NUM = 4 ANGLE = 180 }
			bto_spawn_cluster_home = { NUM = 5 ANGLE = 240 }
			bto_spawn_cluster_home = { NUM = 6 ANGLE = 300 }
			bto_link_cluster_gateway = yes
		}

		# Species, empire, colony, deposits, pops, starbase, neighbours - per slot.
		event_target:fed_planet_1 = { bto_setup_fed_member = { NUM = 1 } }
		event_target:fed_planet_2 = { bto_setup_fed_member = { NUM = 2 } }
		event_target:fed_planet_3 = { bto_setup_fed_member = { NUM = 3 } }
		event_target:fed_planet_4 = { bto_setup_fed_member = { NUM = 4 } }
		event_target:fed_planet_5 = { bto_setup_fed_member = { NUM = 5 } }
		if = {
			limit = { has_origin = origin_clan_confederation }
			event_target:fed_planet_6 = { bto_setup_fed_member = { NUM = 6 } }
		}

		# Leaders, starting ships, federation join, name randomisation.
		event_target:fed_member_1 = { bto_finalize_fed_member = yes }
		event_target:fed_member_2 = { bto_finalize_fed_member = yes }
		event_target:fed_member_3 = { bto_finalize_fed_member = yes }
		event_target:fed_member_4 = { bto_finalize_fed_member = yes }
		event_target:fed_member_5 = { bto_finalize_fed_member = yes }
		if = {
			limit = { has_origin = origin_clan_confederation }
			event_target:fed_member_6 = { bto_finalize_fed_member = yes }
		}

		# Opinion and Federation laws
		if = {
			limit = { has_origin = origin_stellar_union }
			add_opinion_modifier = { who = event_target:fed_member_1 modifier = opinion_common_ground }
			add_opinion_modifier = { who = event_target:fed_member_2 modifier = opinion_common_ground }
			add_opinion_modifier = { who = event_target:fed_member_3 modifier = opinion_common_ground }
			add_opinion_modifier = { who = event_target:fed_member_4 modifier = opinion_common_ground }
			add_opinion_modifier = { who = event_target:fed_member_5 modifier = opinion_common_ground }
			federation = {
				set_federation_type = hegemony_federation
				set_federation_leader = root
				set_name = random
				set_federation_law = succession_term_years_20
				add_federation_experience = 600
				add_cohesion = 100
			}
		}
		if = {
			limit = { has_origin = origin_clan_confederation }
			federation = {
				set_federation_type = hegemony_federation
				set_federation_leader = root
				set_name = random
				set_federation_law = succession_term_years_20
				add_federation_experience = 600
				add_cohesion = 100
			}
		}

		# Lock cohesion and inter-member relations at maximum once the
		# federation has settled.
		country_event = { id = bto_origin.56 days = 3 }
	}
}

# Maximises federation cohesion and the mutual relations of every member.
# Fired by bto_origin.55 for the player and, independently, for the rival.
country_event = {
	id = bto_origin.56
	hide_window = yes
	is_triggered_only = yes

	immediate = {
		if = {
			limit = { has_federation = yes }
			federation = {
				add_cohesion = 100
				# Max mutual opinion and trust between every pair of members.
				every_member = {
					save_event_target_as = bto_unity_member
					federation = {
						every_member = {
							limit = { NOT = { is_same_value = event_target:bto_unity_member } }
							add_trust = {
								who = event_target:bto_unity_member
								amount = 100
							}
							add_opinion_modifier = {
								who = event_target:bto_unity_member
								modifier = opinion_bto_federation_unity
							}
						}
					}
				}
			}
		}
	}
}
