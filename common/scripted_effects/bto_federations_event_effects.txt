### BTO Galaxy Layout Helpers

# All home spawners use the vanilla custom starting initialisers, the same
# system templates spawn_random_home_system draws from. They additionally pin
# orientation angle + distance so the realms form a deliberate layout.

# Called in system scope (the leader's capital). Spawns one Inner Sphere ring
# member at a fixed angle around the leader and registers it as fed_planet_$NUM$.
bto_spawn_ring_home = {
	spawn_system = {
		min_distance = @bto_ring_distance
		max_distance = @bto_ring_distance
		min_orientation_angle = $ANGLE$
		max_orientation_angle = $ANGLE$
		hyperlane = yes
		initializer = custom_starting_init_01
	}
	last_created_system = {
		set_star_flag = bto_fed_origin_used
		federation_origin_planet_setup = { NUM = $NUM$ }
	}
}

# Called in country scope. Spawns the first Clan home far beyond the galactic
# rim (no_scope anchors on the galactic core) as an isolated system, saved as
# event_target:bto_cluster_anchor and registered as fed_planet_$NUM$.
bto_spawn_cluster_anchor = {
	no_scope = {
		spawn_system = {
			min_distance = @bto_cluster_distance
			max_distance = @bto_cluster_distance
			min_orientation_angle = 0
			max_orientation_angle = 0
			hyperlane = no
			initializer = custom_starting_init_01
		}
	}
	last_created_system = {
		set_star_flag = bto_fed_origin_used
		save_event_target_as = bto_cluster_anchor
		federation_origin_planet_setup = { NUM = $NUM$ }
	}
}

# Called in country scope. Spawns one more Clan home around the cluster anchor,
# wires a hyperlane back to the anchor (keeping the cluster connected yet sealed
# from the galaxy) and registers it as fed_planet_$NUM$.
bto_spawn_cluster_home = {
	event_target:bto_cluster_anchor = {
		spawn_system = {
			min_distance = @bto_cluster_spread
			max_distance = @bto_cluster_spread
			min_orientation_angle = $ANGLE$
			max_orientation_angle = $ANGLE$
			hyperlane = no
			initializer = custom_starting_init_01
		}
	}
	last_created_system = {
		set_star_flag = bto_fed_origin_used
		add_hyperlane = { from = this to = event_target:bto_cluster_anchor }
		federation_origin_planet_setup = { NUM = $NUM$ }
	}
}

# Called in a member's home system scope. Claims two neighbour systems for the
# member ($OWNER$) via outposts, each hyperlinked only to the home so a Clan
# member's territory stays inside the sealed cluster.
bto_claim_neighbors = {
	save_event_target_as = bto_neighbor_home
	spawn_system = {
		min_distance = @bto_neighbor_distance
		max_distance = @bto_neighbor_distance
		min_orientation_angle = 45
		max_orientation_angle = 45
		hyperlane = no
		initializer = custom_starting_init_05
	}
	last_created_system = {
		add_hyperlane = { from = this to = event_target:bto_neighbor_home }
		create_starbase = { size = starbase_outpost owner = $OWNER$ }
	}
	spawn_system = {
		min_distance = @bto_neighbor_distance
		max_distance = @bto_neighbor_distance
		min_orientation_angle = 315
		max_orientation_angle = 315
		hyperlane = no
		initializer = custom_starting_init_05
	}
	last_created_system = {
		add_hyperlane = { from = this to = event_target:bto_neighbor_home }
		create_starbase = { size = starbase_outpost owner = $OWNER$ }
	}
}

# Called in country scope (the Clan leader). Places a restored gateway in the
# cluster and one at the leader's capital; gateways auto-network, so this is the
# Clans' only link to the galaxy.
bto_link_cluster_gateway = {
	event_target:bto_cluster_anchor = {
		spawn_megastructure = {
			type = "gateway_restored"
			coords_from = this
		}
	}
	capital_scope = {
		solar_system = {
			spawn_megastructure = {
				type = "gateway_restored"
				coords_from = this
			}
		}
	}
}

### BTO Setup Helpers

# Called from planet scope (event_target:fed_planet_$NUM$).
# Saves the founder's species for the member, spawns the member empire,
# colonises the planet, builds deposits/pops/starbase, claims neighbour systems.
bto_setup_fed_member = {
	solar_system = { generate_home_system_resources = yes }
	# Every federation member shares the founder's (root's) species.
	root = { species = { save_event_target_as = fed_planet_$NUM$_species } }
	if = {
		limit = { root = { has_origin = origin_stellar_union } }
		create_stellar_union_member = { NUM = $NUM$ }
	}
	if = {
		limit = { root = { has_origin = origin_clan_confederation } }
		create_clan_confederation_member = { NUM = $NUM$ }
	}
	create_colony = { owner = event_target:fed_member_$NUM$ }
	generate_start_deposits_and_blockers = yes
	generate_start_buildings_and_districts = yes
	owner = {
		every_owned_planet = { remove_building = building_colony_shelter }
	}
	generate_start_pops = yes
	event_target:fed_member_$NUM$ = {
		country_event = { id = game_start.8 }
		give_starting_resources_effect = yes
		give_technology = { tech = tech_federation_code message = no }
	}
	solar_system = {
		create_starbase = {
			size = starbase_starport
			module = shipyard
			building = crew_quarters
			owner = event_target:fed_member_$NUM$
			effect = {
				set_starbase_module = { slot = 1 module = trading_hub }
				set_starbase_module = { slot = 2 module = shipyard }
			}
		}
	}
	# Two neighbour systems per realm (home + 2 = a small starting territory).
	solar_system = {
		bto_claim_neighbors = { OWNER = event_target:fed_member_$NUM$ }
	}
}

# Called from country scope (event_target:fed_member_N).
# Leaders, ships, federation join, name randomisation.
bto_finalize_fed_member = {
	create_starting_leaders = yes
	country_event = { id = game_start.9 }
	country_event = { id = game_start.33 days = 1 } # needs delay for system ownership to settle
	country_event = { id = origin.58 days = 1 } # delayed for federation assignment to settle
	join_alliance = {
		who = root
		override_requirements = yes
	}
	set_name = random # Fixes some naming issues
}

# Spawns a distant rival empire with the opposing BTO origin.
# Called from country scope (root = the BTO-origin empire).
# Parameters: ORIGIN, AUTH, CIVIC1, CIVIC2
bto_spawn_rival_empire = {
	# Rival shares the player's species (same lineage, different political path)
	species = { save_event_target_as = bto_rival_species }
	# Pick a distant system; min_distance scales with galaxy size.
	if = {
		limit = { galaxy_size = tiny }
		random_system = {
			limit = {
				has_owner = no
				NOT = { has_star_flag = empire_cluster }
				NOT = { has_star_flag = bto_fed_origin_used }
				distance = { source = root.capital_scope  min_distance >= 50 }
			}
			spawn_random_home_system = yes
			last_created_system = { federation_origin_planet_setup = { NUM = 99 } }
		}
	}
	else_if = {
		limit = { galaxy_size = small }
		random_system = {
			limit = {
				has_owner = no
				NOT = { has_star_flag = empire_cluster }
				NOT = { has_star_flag = bto_fed_origin_used }
				distance = { source = root.capital_scope  min_distance >= 80 }
			}
			spawn_random_home_system = yes
			last_created_system = { federation_origin_planet_setup = { NUM = 99 } }
		}
	}
	else_if = {
		limit = { galaxy_size = medium }
		random_system = {
			limit = {
				has_owner = no
				NOT = { has_star_flag = empire_cluster }
				NOT = { has_star_flag = bto_fed_origin_used }
				distance = { source = root.capital_scope  min_distance >= 100 }
			}
			spawn_random_home_system = yes
			last_created_system = { federation_origin_planet_setup = { NUM = 99 } }
		}
	}
	else_if = {
		limit = { galaxy_size = large }
		random_system = {
			limit = {
				has_owner = no
				NOT = { has_star_flag = empire_cluster }
				NOT = { has_star_flag = bto_fed_origin_used }
				distance = { source = root.capital_scope  min_distance >= 130 }
			}
			spawn_random_home_system = yes
			last_created_system = { federation_origin_planet_setup = { NUM = 99 } }
		}
	}
	else = {
		# huge or any unrecognised size
		random_system = {
			limit = {
				has_owner = no
				NOT = { has_star_flag = empire_cluster }
				NOT = { has_star_flag = bto_fed_origin_used }
				distance = { source = root.capital_scope  min_distance >= 150 }
			}
			spawn_random_home_system = yes
			last_created_system = { federation_origin_planet_setup = { NUM = 99 } }
		}
	}
	event_target:fed_planet_99 = {
		create_country = {
			name = random
			name_list = random
			species = event_target:bto_rival_species
			type = default
			origin = origin_default
			ignore_initial_colony_error = yes
			day_zero_contact = no
			ethos = {
				ethic = "ethic_fanatic_militarist"
				ethic = "ethic_authoritarian"
			}
			authority = $AUTH$
			civics = {
				civic = $CIVIC1$
				civic = $CIVIC2$
			}
			effect = {
				save_event_target_as = bto_rival_empire
				establish_communications_no_message = root
				set_country_flag = bto_is_rival_empire
			}
		}
		create_colony = { owner = event_target:bto_rival_empire }
		generate_start_deposits_and_blockers = yes
		generate_start_buildings_and_districts = yes
		if = {
			limit = { owner = { any_owned_planet = { has_building = building_hive_capital } } }
			owner = { every_owned_planet = { remove_building = building_hive_capital } }
		}
		else = {
			owner = { every_owned_planet = { remove_building = building_colony_shelter } }
		}
		generate_start_pops = yes
		solar_system = { generate_home_system_resources = yes }
		solar_system = {
			create_starbase = {
				size = starbase_starport
				module = shipyard
				building = crew_quarters
				owner = event_target:bto_rival_empire
				effect = {
					set_starbase_module = { slot = 1 module = trading_hub }
					set_starbase_module = { slot = 2 module = shipyard }
				}
			}
		}
	}
	event_target:bto_rival_empire = {
		# Assign the BTO origin now that civics exist (the origin's possible
		# requires civic_warrior_culture, which create_country sets after the
		# origin would be validated — so it cannot be passed to create_country).
		set_origin = $ORIGIN$
		country_event = { id = game_start.8 }
		give_starting_resources_effect = yes
		give_technology = { tech = tech_federation_code message = no }
		create_starting_leaders = yes
		country_event = { id = game_start.9 }
		country_event = { id = bto_origin.55 }
	}
}
